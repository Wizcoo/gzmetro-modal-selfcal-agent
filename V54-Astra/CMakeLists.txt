cmake_minimum_required(VERSION 3.16)
project(ModalSelfCalAstraCurrent LANGUAGES CXX)
set(ASTRA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/round8_release_111315/source")
add_library(modal_selfcal_astra STATIC "${ASTRA_SOURCE_DIR}/ModalSelfCal.cpp")
target_include_directories(modal_selfcal_astra PUBLIC "${ASTRA_SOURCE_DIR}")
target_compile_features(modal_selfcal_astra PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(modal_selfcal_astra PRIVATE /utf-8 /W4)
else()
  target_compile_options(modal_selfcal_astra PRIVATE -Wall -Wextra)
endif()
option(ASTRA_BUILD_STATE_VERIFIER "Build existing opt-in verifier; does not run it" OFF)
if(ASTRA_BUILD_STATE_VERIFIER)
  add_executable(astra_verify_release round8_release_111315/tools/verify_release.cpp)
  target_link_libraries(astra_verify_release PRIVATE modal_selfcal_astra)
endif()
